Private Sub UserForm_Initialize()

Private Sub LoadSettings()
    Dim i As Integer
    For i = 1 To 8
        Me.Controls("chk1_" & i).Value = (CfgGet("c1_" & i, IIf(i <= 5 Or i = 8, "1", "0")) = "1")
        Me.Controls("txt1_" & i).Text = CfgGet("t1_" & i, CStr(i))
    Next i
    For i = 1 To 6
        Me.Controls("chkM_" & i).Value = (CfgGet("tpl_" & i, "1") = "1")
        Me.Controls("txtN_" & i).Text = CfgGet("tN_" & i, CStr(i))
    Next i
    Me.chkTong.Value = (CfgGet("tong", "1") = "1")
    Me.chkOpen.Value = (CfgGet("open", "1") = "1")
    Me.chkSub.Value = (CfgGet("sub", "0") = "1")
    Me.chkTenHM.Value = (CfgGet("tenhm", "0") = "1")
    Me.txtOut.Text = CfgGet("out", "")
    Me.txtFolder.Text = CfgGet("folder", "")
    If CfgGet("xk", "TACH") = "GHEP" Then Me.optGhep.Value = True Else Me.optTach.Value = True
End Sub

Private Sub SaveSettings()
    Dim i As Integer
    ' Dong bi mo (sheet khong co trong file dang chon) giu nguyen gia tri da luu
    For i = 1 To 8
        If Me.Controls("chk1_" & i).Enabled Then
            CfgSet "c1_" & i, IIf(Me.Controls("chk1_" & i).Value, "1", "0")
            CfgSet "t1_" & i, Trim$(Me.Controls("txt1_" & i).Text)
        End If
    Next i
    For i = 1 To 6
        If Me.Controls("chkM_" & i).Enabled Then
            CfgSet "tpl_" & i, IIf(Me.Controls("chkM_" & i).Value, "1", "0")
            CfgSet "tN_" & i, Trim$(Me.Controls("txtN_" & i).Text)
        End If
    Next i
    CfgSet "tong", IIf(Me.chkTong.Value, "1", "0")
    CfgSet "open", IIf(Me.chkOpen.Value, "1", "0")
    CfgSet "sub", IIf(Me.chkSub.Value, "1", "0")
    CfgSet "tenhm", IIf(Me.chkTenHM.Value, "1", "0")
    CfgSet "out", Trim$(Me.txtOut.Text)
    CfgSet "folder", Trim$(Me.txtFolder.Text)
    CfgSet "xk", IIf(Me.optGhep.Value, "GHEP", "TACH")
End Sub

' preferActive = True: chon workbook dang active (neu co trong danh sach)
Private Sub FillWorkbookList(Optional preferActive As Boolean = False)
    Dim wb As Workbook, keep As Workbook, pick As Integer
    Set keep = CurWb()
    mLoading = True
    Me.cboFile.Clear
    Set mWbs = New Collection
    pick = -1
    For Each wb In Application.Workbooks
        If Not (wb Is ThisWorkbook) And Not wb.IsAddin Then
            If WbVisible(wb) Then
                Me.cboFile.AddItem wb.Name
                mWbs.Add wb
                If preferActive Or keep Is Nothing Then
                    If wb Is ActiveWorkbook Then pick = Me.cboFile.ListCount - 1
                ElseIf wb Is keep Then
                    pick = Me.cboFile.ListCount - 1
                End If
            End If
        End If
    Next wb
    If pick < 0 And Me.cboFile.ListCount > 0 Then pick = 0
    If pick >= 0 Then Me.cboFile.ListIndex = pick
    mLoading = False
    ' Chi nhan dien lai khi doi sang workbook khac (nhan dien file lon ton thoi gian)
    If Not (CurWb() Is mCurWb) Or mCurWb Is Nothing Then
        DetectCurrent
        RefreshMode
    End If
End Sub

Private Function WbVisible(wb As Workbook) As Boolean
    Dim w As Window
    On Error Resume Next
    For Each w In wb.Windows
        If w.Visible Then WbVisible = True: Exit Function
    Next w
End Function

Private Function CurWb() As Workbook
    If Me.cboFile.ListIndex < 0 Then Exit Function
    On Error Resume Next
    Set CurWb = mWbs(Me.cboFile.ListIndex + 1)
    If Not CurWb Is Nothing Then
        If Len(CurWb.Name) = 0 Then Set CurWb = Nothing   ' workbook da dong
    End If
    On Error GoTo 0
End Function

Private Sub DetectCurrent()
    Dim wb As Workbook, names() As String, r As Integer
    mFound = 0: mNHM = 0
    ReDim mEx1(1 To 8): ReDim mEx2(1 To 6)
    Set wb = CurWb()
    Set mCurWb = wb
    If Not wb Is Nothing Then
        Application.Cursor = xlWait
        mFound = API_Detect(wb, mNHM, names, mEx1, mEx2)
        Application.Cursor = xlDefault
    End If
    ' dong hang muc
    mLoading = True
    For r = 1 To ROWS_MAX
        If r <= mNHM Then
            Me.fraHM.Controls("lblH_" & r).Caption = r & ".  " & names(r)
            Me.fraHM.Controls("lblH_" & r).ControlTipText = names(r)
        End If
    Next r
    ApplyMasterToRows
    mLoading = False
End Sub

Private Sub ScanFolder()
    mNFiles = 0
    mFiles = Empty
    Dim f As String: f = Trim$(Me.txtFolder.Text)
    If Len(f) = 0 Then Exit Sub
    mFiles = API_CollectFiles(f, CBool(Me.chkSub.Value), mNFiles)
End Sub

                If k = 6 Then v = v And (r = 1)
                HChk(r, k).Value = v

        End If
    Next k
End Sub

' Bam ten hang muc: dang co sheet tick -> bo het ; dang trong -> tick theo dong mau
Private Sub ToggleRow(r As Integer)

    Dim k As Integer, anyOn As Boolean
    For k = 1 To 6
        If HChk(r, k).Enabled And HChk(r, k).Value Then anyOn = True
    Next k
    Dim v As Boolean
    For k = 1 To 6

        End If
    Next k
End Sub

Private Sub RefreshMode()
    Dim folderMode As Boolean: folderMode = Me.optFolder.Value
    Dim i As Integer, r As Integer, k As Integer, on1 As Boolean, onN As Boolean

    If folderMode Then
        on1 = True: onN = True
        If Len(Trim$(Me.txtFolder.Text)) = 0 Then
            SetDetect "Chưa chọn thư mục.", ClrTextDim()
        ElseIf mNFiles = 0 Then
            SetDetect "Không có file Excel nào trong thư mục này.", ClrRed()
        Else
            SetDetect "Thư mục: " & mNFiles & " file Excel  —  mỗi file tự nhận diện 1 hay nhiều hạng mục.", ClrGreen()
        End If
    Else
        If CurWb() Is Nothing Then
            SetDetect "Không có file nào đang mở. Bấm 'Mở file...' hoặc chọn 'Cả thư mục'.", ClrTextDim()
        ElseIf mFound = 0 Then
            SetDetect "Không nhận ra sheet dự toán G8 / ESCON trong file này.", ClrRed()
        ElseIf mNHM >= 2 Then
            onN = True
            SetDetect "Nhận diện: " & mNHM & " HẠNG MỤC  —  tìm thấy " & mFound & "/6 sheet.", ClrGreen()
        Else
            on1 = True
            SetDetect "Nhận diện: 1 HẠNG MỤC  —  tìm thấy " & mFound & "/8 sheet.", ClrOrange()
        End If
    End If

    ' panel 1 hang muc
    Me.lblP1.ForeColor = IIf(on1, ClrOrange(), ClrTextOff())
    For i = 1 To 8
        Dim r1 As Boolean: r1 = on1
        If on1 And Not folderMode Then r1 = mEx1(i)
        Me.Controls("chk1_" & i).Enabled = r1
        Me.Controls("txt1_" & i).Enabled = r1
        Me.Controls("lbl1_" & i).ForeColor = IIf(r1, ClrText(), ClrTextOff())
        Me.Controls("lbl1_" & i).Caption = Name1(i) & IIf(on1 And Not r1, "   (không có)", "")
    Next i
    SetBtn "btnAll", on1: SetBtn "btnNone", on1: SetBtn "btn5", on1

    ' panel nhieu hang muc
    Me.lblP2.ForeColor = IIf(onN, ClrGreen(), ClrTextOff())
    Me.lblP2.Caption = IIf(onN And Not folderMode, mNHM & " HẠNG MỤC  —  tick sheet cho từng hạng mục", _
                                                   "NHIỀU HẠNG MỤC  —  tick sheet cho từng hạng mục")
    Me.lblMaster.Caption = IIf(folderMode, "Mẫu cho mọi file", "Tất cả hạng mục")
    Me.lblMaster.ForeColor = IIf(onN, ClrText(), ClrTextOff())
    For k = 1 To 6
        Dim colOn As Boolean: colOn = onN
        If onN And Not folderMode Then colOn = mEx2(k)
        Me.Controls("chkM_" & k).Enabled = colOn
        Me.Controls("txtN_" & k).Enabled = colOn
        Me.Controls("lblC2_" & k).ForeColor = IIf(colOn, ClrTextDim(), ClrTextOff())
    Next k

    Dim nRows As Integer
    If onN And Not folderMode Then nRows = mNHM
    For r = 1 To ROWS_MAX
        Dim vis As Boolean: vis = (r <= nRows)
        Me.fraHM.Controls("lblH_" & r).Visible = vis
        If r Mod 2 = 0 Then Me.fraHM.Controls("rowH_" & r).Visible = vis
        For k = 1 To 6
            HChk(r, k).Visible = vis
            HChk(r, k).Enabled = vis And mEx2(k)
        Next k
    Next r
    Me.fraHM.Controls("lblFolderNote").Visible = folderMode
    Me.fraHM.ScrollHeight = IIf(nRows * ROW_H > Me.fraHM.Height, nRows * ROW_H + 2, Me.fraHM.Height)
    Me.fraHM.ScrollTop = 0

    SetOpt "optTach", onN: SetOpt "optGhep", onN
    SetOpt "chkTenHM", onN And Me.optTach.Value

    ' nut thao tac
    Dim canRun As Boolean, isN As Boolean
    isN = (mNHM >= 2)
    If folderMode Then canRun = (mNFiles > 0) Else canRun = (mFound > 0)
    SetBtn "btnAlign", canRun And Not folderMode And Not isN
    SetBtn "btnPreview", canRun And Not folderMode And Not isN
    SetBtn "btnPrint", canRun And (folderMode Or Not isN)
    SetBtn "btnPDF", canRun
    If folderMode And mNFiles > 0 Then
        Me.btnPDF_t.Caption = "XUẤT PDF  (" & mNFiles & " file)"
    ElseIf Not folderMode And isN Then
        Me.btnPDF_t.Caption = IIf(Me.optGhep.Value, "XUẤT PDF  (ghép 1 file)", "XUẤT PDF  (mỗi HM 1 file)")
    Else
        Me.btnPDF_t.Caption = "XUẤT PDF"
    End If
End Sub

' ============================================================================

Public Sub OnBtn(nm As String)
    If mBusy Then
        If nm = "btnClose" Then mStop = True: AppendLog "Đang dừng sau file hiện tại..."
        Exit Sub
    End If
    If Left$(nm, 4) = "row:" Then ToggleRow CInt(Mid$(nm, 5)): Exit Sub
    If Me.Controls(nm).Tag <> "1" Then Exit Sub

    Select Case nm
        Case "btnOpenFile": DoOpenFile
        Case "btnRefresh":  FillWorkbookList True: ScanFolder: RefreshMode
        Case "btnFolder":   DoPickFolder
        Case "btnAll":      SetAll1 0
        Case "btnNone":     SetAll1 -1
        Case "btn5":        SetAll1 5
        Case "btnOut":      DoPickOut
        Case "btnOutClr":   Me.txtOut.Text = ""
        Case "btnAlign":    DoRun "ALIGN"
        Case "btnPreview":  DoRun "PREVIEW"
        Case "btnPrint":    DoRun "PRINT"
        Case "btnPDF":      DoRun "PDF"
        Case "btnClose":    CloseForm
    End Select
End Sub

Private Sub DoOpenFile()
    Dim p As Variant, wb As Workbook, w As Workbook
    p = Application.GetOpenFilename("Excel (*.xls;*.xlsx;*.xlsm),*.xls;*.xlsx;*.xlsm", , _
                                    "Chọn file dự toán", , False)
    If VarType(p) = vbBoolean Then Exit Sub
    For Each w In Application.Workbooks
        If StrComp(w.FullName, CStr(p), vbTextCompare) = 0 Then Set wb = w: Exit For
    Next w
    If wb Is Nothing Then
        On Error Resume Next
        Set wb = Workbooks.Open(CStr(p), UpdateLinks:=0)
        On Error GoTo 0
    End If
    If wb Is Nothing Then AppendLog "Không mở được: " & CStr(p): Exit Sub
    wb.Activate
    mLoading = True: Me.optFile.Value = True: mLoading = False
    FillWorkbookList True
    AppendLog "Đã mở: " & wb.Name
End Sub

Private Function PickFolder(title As String, start As String) As String
    Dim fd As Object
    Set fd = Application.FileDialog(4)          ' msoFileDialogFolderPicker
    fd.Title = title
    fd.AllowMultiSelect = False
    If Len(start) > 0 Then fd.InitialFileName = start & IIf(Right$(start, 1) = "\", "", "\")
    If fd.Show = -1 Then PickFolder = fd.SelectedItems(1)
End Function

Private Sub DoPickFolder()
    Dim f As String
    f = PickFolder("Chọn thư mục chứa các file dự toán", Trim$(Me.txtFolder.Text))
    If Len(f) = 0 Then Exit Sub
    Me.txtFolder.Text = f
    mLoading = True: Me.optFolder.Value = True: mLoading = False
    ScanFolder
    RefreshMode
End Sub

Private Sub DoPickOut()
    Dim f As String
    f = PickFolder("Chọn thư mục lưu PDF", Trim$(Me.txtOut.Text))
    If Len(f) > 0 Then Me.txtOut.Text = f
End Sub

Private Function Harvest() As Boolean
    Harvest = False
    Dim i As Integer, k As Integer, r As Integer, t As String
    For i = 1 To 8
        gSelSheet(i) = Me.Controls("chk1_" & i).Value And Me.Controls("chk1_" & i).Enabled
        gOrder(i) = ReadTT(Me.Controls("txt1_" & i))
    Next i
    For k = 1 To 6
        gTplSel(k) = Me.Controls("chkM_" & k).Value
        gOrderN(k) = ReadTT(Me.Controls("txtN_" & k))
        For r = 1 To MAX_HM
            gSelN(k, r) = False
            If r <= mNHM And r <= ROWS_MAX Then gSelN(k, r) = HChk(r, k).Value And HChk(r, k).Enabled
        Next r
    Next k
    gGiaTongHopInTongCong = Me.chkTong.Value
    gGTH_InTongCong2 = Me.chkTong.Value
    gHM_XuatKieu = IIf(Me.optGhep.Value, "GHEP", "TACH")
    gHM_TenTheoHM = Me.chkTenHM.Value

    t = Trim$(Me.txtOut.Text)
    If Len(t) > 0 Then
        If Not CreateObject("Scripting.FileSystemObject").FolderExists(t) Then
            AppendLog "Thư mục lưu PDF không tồn tại: " & t
            Exit Function
        End If
    End If
    gOutFolder1 = t
    gHM_OutFolder = t
    Harvest = True
End Function

' So thu tu: rong / sai -> 99 (xep cuoi)
Private Function ReadTT(c As Object) As Integer
    Dim t As String: t = Trim$(c.Text)
    If Len(t) > 0 And IsNumeric(t) Then
        ReadTT = CInt(Val(t))
        If ReadTT < 1 Then ReadTT = 99
    Else
        ReadTT = 99
    End If
End Function

Private Function Count1() As Integer
    Dim i As Integer
    For i = 1 To 8
        If gSelSheet(i) Then Count1 = Count1 + 1
    Next i
End Function

Private Function CountN() As Integer
    Dim k As Integer, r As Integer
    For r = 1 To mNHM
        For k = 1 To 6
            If gSelN(k, r) Then CountN = CountN + 1
        Next k
    Next r
End Function

Private Sub DoRun(act As String)
    If Not Harvest() Then Exit Sub
    SaveSettings

    If Me.optFolder.Value Then
        RunFolder act = "PDF"
        Exit Sub
    End If

    Dim wb As Workbook: Set wb = CurWb()
    If wb Is Nothing Then AppendLog "Chưa chọn file (hoặc file đã đóng) — bấm Làm mới.": Exit Sub

    ' Form modeless: file co the da bi sua tu luc nhan dien -> nhan dien lai, so khop so hang muc
    Dim nHM As Integer, names() As String, e1() As Boolean, e2() As Boolean
    If API_Detect(wb, nHM, names, e1, e2) = 0 Or nHM <> mNHM Then
        AppendLog "File đã thay đổi so với lúc nhận diện — đã nhận diện lại, kiểm tra rồi bấm lại."
        DetectCurrent: RefreshMode
        Exit Sub
    End If

    If mNHM >= 2 Then
        If CountN() = 0 Then AppendLog "Chưa tick sheet nào cho hạng mục nào.": Exit Sub
    Else
        If Count1() = 0 Then AppendLog "Chưa tick sheet nào để in.": Exit Sub
    End If

    Dim msg As String, ok As Boolean
    If act = "PREVIEW" Then
        ' Xem truoc can giao dien Excel: an form, xem, roi hien lai
        Me.Hide
        ok = API_Run1HM(wb, "PREVIEW", msg)
        Me.Show vbModeless
        If Not ok Then AppendLog "LỖI  " & wb.Name & "  —  " & msg
        Exit Sub
    End If

    BeginBusy "Đang xử lý " & wb.Name & " ..."
    If mNHM >= 2 Then
        ok = API_RunNHM(wb, msg)
    Else
        ok = API_Run1HM(wb, act, msg)
    End If
    EndBusy
    AppendLog IIf(ok, "OK  ", "LỖI  ") & wb.Name & "  —  " & msg
    If ok And act = "PDF" Then OpenOutFolder OutFolderOf(wb.Path)
    RefreshMode
End Sub

Private Sub RunFolder(isPDF As Boolean)
    ScanFolder
    If mNFiles = 0 Then AppendLog "Không có file Excel nào trong thư mục.": RefreshMode: Exit Sub

    Dim i As Long, k As Integer, msg As String, ok As Boolean
    Dim n1 As Long, n2 As Long, nErr As Long, nSkip As Long, errText As String
    Dim root As String: root = Trim$(Me.txtFolder.Text)

    AppendLog "Bắt đầu " & IIf(isPDF, "xuất PDF", "in máy in") & " " & mNFiles & " file trong: " & root
    BeginBusy ""
    For i = 1 To mNFiles
        If mStop Then AppendLog "Đã dừng theo yêu cầu.": Exit For
        SetDetect "Đang xử lý " & i & "/" & mNFiles & ":  " & ShortName(CStr(mFiles(i))), ClrOrange()
        Me.Repaint
        DoEvents
        ok = API_ProcessFile(CStr(mFiles(i)), isPDF, k, msg)
        If ok Then
            If k = 2 Then n2 = n2 + 1 Else n1 = n1 + 1
        ElseIf k = 0 Then
            nSkip = nSkip + 1
        Else
            nErr = nErr + 1
            errText = errText & vbCrLf & "  " & CStr(mFiles(i)) & vbCrLf & "      " & msg
        End If
        AppendLog "[" & i & "/" & mNFiles & "] " & IIf(ok, "OK  ", IIf(k = 0, "BỎ QUA  ", "LỖI  ")) & _
                  ShortName(CStr(mFiles(i))) & "  —  " & msg
    Next i
    EndBusy

    Dim sum As String
    sum = "XONG: 1 hạng mục " & n1 & " file, nhiều hạng mục " & n2 & " file, lỗi " & nErr & ", bỏ qua " & nSkip & "."
    If nErr > 0 Then
        Dim lp As String
        lp = API_WriteLog(root, "KET QUA IN THU MUC - " & Format(Now, "yyyy-mm-dd hh:nn:ss") & vbCrLf & _
                                "Thu muc: " & root & vbCrLf & _
                                "1HM: " & n1 & " | nhieu HM: " & n2 & " | Loi: " & nErr & vbCrLf & _
                                "==============================" & errText)
        If Len(lp) > 0 Then sum = sum & "  Danh sách lỗi: " & lp
    End If
    AppendLog sum
    If isPDF And (n1 + n2) > 0 Then OpenOutFolder OutFolderOf(root)
    FillWorkbookList
    RefreshMode
End Sub

Private Function ShortName(p As String) As String
    Dim i As Long: i = InStrRev(p, "\")
    If i > 0 Then ShortName = Mid$(p, i + 1) Else ShortName = p
End Function

Private Function OutFolderOf(defaultFolder As String) As String
    If Len(Trim$(Me.txtOut.Text)) > 0 Then OutFolderOf = Trim$(Me.txtOut.Text) Else OutFolderOf = defaultFolder
End Function

Private Sub OpenOutFolder(f As String)
    If Not Me.chkOpen.Value Or Len(f) = 0 Then Exit Sub
    On Error Resume Next
    CreateObject("Shell.Application").Open CVar(f)
    On Error GoTo 0
End Sub

Private Sub BeginBusy(status As String)
    mBusy = True: mStop = False
    ResetHover
    If Len(status) > 0 Then SetDetect status, ClrOrange()
    Me.btnClose_t.Caption = "Dừng"
    Me.MousePointer = 11            ' fmMousePointerHourGlass
    Me.Repaint
    DoEvents
End Sub

Private Sub EndBusy()
    mBusy = False
    Me.btnClose_t.Caption = "Đóng"
    Me.MousePointer = 0
End Sub

' Nhat ky: dong moi nhat o tren cung
Private Sub AppendLog(s As String)
    mLog = Format$(Now, "hh:nn:ss") & "   " & s & vbCrLf & mLog
    If Len(mLog) > 20000 Then mLog = Left$(mLog, 20000)
    Me.txtLog.Text = mLog
    Me.txtLog.SelStart = 0
    Me.Repaint
End Sub

Private Sub UserForm_QueryClose(Cancel As Integer, CloseMode As Integer)
    If CloseMode = 0 Then
        If mBusy Then
            Cancel = True
            mStop = True
        Else
            SaveSettings
            Set modInAnMain.gFrm = Nothing
        End If
    End If
End Sub
